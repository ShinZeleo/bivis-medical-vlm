def wilson_ci(p, n, z=1.96):
    if n == 0:
        return [0.0, 0.0]
    denom = 1 + z**2 / n
    centre = (p + z**2 / (2 * n)) / denom
    delta = z * ((p * (1 - p) / n + z**2 / (4 * n**2)) ** 0.5) / denom
    return [round(max(0.0, centre - delta) * 100, 2), round(min(1.0, centre + delta) * 100, 2)]

def report_subset_metrics(df_sub, label):
    n = len(df_sub)
    acc_en = df_sub["correct_en"].mean()
    acc_id = df_sub["correct_id"].mean()
    gap = abs(acc_en - acc_id)
    flip = df_sub["flip"].mean()
    joint = df_sub["joint_correct"].mean()
    mean_conf_en = df_sub["confidence_en"].mean()
    mean_conf_id = df_sub["confidence_id"].mean()
    val_en = df_sub["valid_format_en"].sum()
    val_id = df_sub["valid_format_id"].sum()

    ci_en = wilson_ci(acc_en, n)
    ci_id = wilson_ci(acc_id, n)
    ci_flip = wilson_ci(flip, n)
    ci_joint = wilson_ci(joint, n)

    print("=" * 68)
    print(f"📊 {label.upper()} (N = {n}) - MODEL: {STRONG_MODEL_ID}")
    print("=" * 68)
    print(f"1. English Accuracy      (Acc_EN)      : {acc_en:.4f} ({acc_en*100:.2f}%)  [95% CI: {ci_en[0]}% - {ci_en[1]}%]")
    print(f"2. Indonesian Accuracy   (Acc_ID)      : {acc_id:.4f} ({acc_id*100:.2f}%)  [95% CI: {ci_id[0]}% - {ci_id[1]}%]")
    print(f"3. Language Gap          (|EN - ID|)   : {gap:.4f} ({gap*100:.2f} pts)")
    print(f"4. Cross-Lingual Flip    (FlipRate)    : {flip:.4f} ({flip*100:.2f}%)  [95% CI: {ci_flip[0]}% - {ci_flip[1]}%]")
    print(f"5. Joint Correctness     (JointAcc)    : {joint:.4f} ({joint*100:.2f}%)  [95% CI: {ci_joint[0]}% - {ci_joint[1]}%]")
    print(f"6. Mean Confidence       (EN / ID)     : {mean_conf_en:.4f} / {mean_conf_id:.4f}")
    print(f"7. Valid Format Count    (EN / ID)     : {val_en}/{n} ({val_en/n*100:.1f}%) / {val_id}/{n} ({val_id/n*100:.1f}%)")
    return {
        "n": n, "acc_en": acc_en, "acc_id": acc_id, "gap": gap, "flip": flip, "joint": joint,
        "ci_en": ci_en, "ci_id": ci_id, "ci_flip": ci_flip, "ci_joint": ci_joint
    }
